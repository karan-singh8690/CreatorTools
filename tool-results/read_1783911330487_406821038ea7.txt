    79→// ─── Client-side SHA256 (Web Crypto API) ─────────────────────────────────────
    80→
    81→export async function computeClientHash(file: File): Promise<string> {
    82→  const buffer = await file.arrayBuffer()
    83→  const hashBuffer = await crypto.subtle.digest('SHA-256', buffer)
    84→  const hashArray = Array.from(new Uint8Array(hashBuffer))
    85→  return hashArray.map((b) => b.toString(16).padStart(2, '0')).join('')
    86→}

    88→// ─── Upload with real progress using XMLHttpRequest ──────────────────────────
    89→
    90→function uploadWithProgress(
    91→  file: File,
    92→  onProgress: (progress: number) => void
    93→): Promise<{ response: Response; data: any }> {
    94→  return new Promise((resolve, reject) => {
    95→    const xhr = new XMLHttpRequest()
    96→    const formData = new FormData()
    97→    formData.append('file', file)
    98→
    99→    xhr.upload.addEventListener('progress', (event) => {
   100→      if (event.lengthComputable) {
   101→        const percent = Math.round((event.loaded / event.total) * 100)
   102→        onProgress(percent)
   103→      }
   104→    })
   105→
   106→    xhr.addEventListener('load', () => {
   107→      const response = new Response(xhr.response, {
   108→        status: xhr.status,
   109→        statusText: xhr.statusText,
   110→        headers: new Headers({
   111→          'Content-Type': xhr.getResponseHeader('Content-Type') || 'application/json',
   112→        }),
   113→      })
   114→
   115→      let data: any
   116→      try {
   117→        data = JSON.parse(xhr.responseText)
   118→      } catch {
   119→        data = { error: 'Invalid response' }
   120→      }
   121→
   122→      resolve({ response, data })
   123→    })
   124→
   125→    xhr.addEventListener('error', () => {
   126→      reject(new Error('Network error during upload'))
   127→    })
   128→
   129→    xhr.addEventListener('abort', () => {
   130→      reject(new Error('Upload was aborted'))
   131→    })
   132→
   133→    xhr.open('POST', '/api/files')
   134→    xhr.responseType = 'text'
   135→    xhr.send(formData)
   136→  })
   137→}

   139→// ─── Chunked Upload for Large Files ──────────────────────────────────────────
   140→
   141→async function chunkedUploadWithProgress(
   142→  file: File,
   143→  fileHash: string,
   144→  onProgress: (progress: number) => void
   145→): Promise<{ data: any }> {
   146→  const totalChunks = Math.ceil(file.size / UPLOAD_LIMITS.CHUNK_SIZE)
   147→
   148→  // Step 1: Init session
   149→  const initResponse = await fetch('/api/files/chunk', {
   150→    method: 'POST',
   151→    headers: { 'Content-Type': 'application/json' },
   152→    body: JSON.stringify({
   153→      action: 'init',
   154→      fileName: file.name,
   155→      fileSize: file.size,
   156→      totalChunks,
   157→      fileHash,
   158→    }),
   159→  })
   160→
   161→  if (!initResponse.ok) {
   162→    const errorData = await initResponse.json().catch(() => ({ error: 'Init failed' }))
   163→    // Handle duplicate detection at init stage
   164→    if (initResponse.status === 409) {
   165→      throw Object.assign(new Error(errorData.error || 'Duplicate file'), {
   166→        code: 'DUPLICATE_FILE',
   167→        duplicate: errorData.duplicate,
   168→      })
   169→    }
   170→    throw new Error(errorData.error || 'Failed to initialize chunked upload')
   171→  }
   172→
   173→  const { sessionId } = await initResponse.json()
   174→
   175→  // Step 2: Upload chunks sequentially
   176→  for (let i = 0; i < totalChunks; i++) {
   177→    const start = i * UPLOAD_LIMITS.CHUNK_SIZE
   178→    const end = Math.min(start + UPLOAD_LIMITS.CHUNK_SIZE, file.size)
   179→    const chunk = file.slice(start, end)
   180→
   181→    const chunkFormData = new FormData()
   182→    chunkFormData.append('sessionId', sessionId)
   183→    chunkFormData.append('chunkIndex', String(i))
   184→    chunkFormData.append('chunk', chunk)
   185→
   186→    const chunkResponse = await fetch('/api/files/chunk', {
   187→      method: 'POST',
   188→      body: chunkFormData,
   189→    })
   190→
   191→    if (!chunkResponse.ok) {
   192→      throw new Error(`Chunk ${i + 1}/${totalChunks} upload failed`)
   193→    }
   194→
   195→    // Update progress: chunk progress + overall progress
   196→    const chunkProgress = ((i + 1) / totalChunks) * 100
   197→    onProgress(Math.round(chunkProgress))
   198→  }
   199→
   200→  // Step 3: Complete upload
   201→  const completeResponse = await fetch('/api/files/chunk', {
   202→    method: 'POST',
   203→    headers: { 'Content-Type': 'application/json' },
   204→    body: JSON.stringify({ action: 'complete', sessionId }),
   205→  })
   206→
   207→  if (!completeResponse.ok) {
   208→    const errorData = await completeResponse.json().catch(() => ({ error: 'Complete failed' }))
   209→    if (completeResponse.status === 409) {
   210→      throw Object.assign(new Error(errorData.error || 'Duplicate file'), {
   211→        code: 'DUPLICATE_FILE',
   212→        duplicate: errorData.duplicate,
   213→      })
   214→    }
   215→    if (completeResponse.status === 422) {
   216→      throw Object.assign(new Error(errorData.error || 'Security threat'), {
   217→        code: 'VIRUS_DETECTED',
   218→        threats: errorData.threats,
   219→      })
   220→    }
   221→    throw new Error(errorData.error || 'Failed to complete chunked upload')
   222→  }
   223→
   224→  const data = await completeResponse.json()
   225→  return { data }
   226→}

   354→
   355→  // ─── Enhanced Upload with Validation, Progress, Duplicate Detection ──────
   356→  uploadFiles: async (files: File[], onProgress?: (progresses: UploadProgress[]) => void) => {
   357→    // Initialize progress for all files
   358→    const progresses: UploadProgress[] = files.map((f) => ({
   359→      fileName: f.name,
   360→      fileSize: f.size,
   361→      progress: 0,
   362→      status: 'validating' as const,
   363→    }))
   364→    onProgress?.(progresses)
   365→
   366→    const uploadedFiles: PdfFile[] = []
   367→
   368→    // ── Client-side pre-validation ────────────────────────────────────────
   369→    for (let i = 0; i < files.length; i++) {
   370→      const file = files[i]
   371→
   372→      // Check file extension
   373→      const ext = file.name.toLowerCase().slice(file.name.lastIndexOf('.'))
   374→      if (!UPLOAD_LIMITS.ALLOWED_EXTENSIONS.includes(ext)) {
   375→        progresses[i] = {
   376→          ...progresses[i],
   377→          status: 'error',
   378→          error: `Invalid file type. Only PDF files are accepted.`,
   379→          errorCode: 'INVALID_EXTENSION',
   380→        }
   381→        onProgress?.([...progresses])
   382→        continue
   383→      }
   384→
   385→      // Check file size
   386→      if (file.size > UPLOAD_LIMITS.MAX_FILE_SIZE) {
   387→        progresses[i] = {
   388→          ...progresses[i],
   389→          status: 'error',
   390→          error: `File size (${formatFileSize(file.size)}) exceeds limit (${formatFileSize(UPLOAD_LIMITS.MAX_FILE_SIZE)})`,
   391→          errorCode: 'SIZE_EXCEEDED',
   392→        }
   393→        onProgress?.([...progresses])
   394→        continue
   395→      }
   396→
   397→      if (file.size < 100) {
   398→        progresses[i] = {
   399→          ...progresses[i],
   400→          status: 'error',
   401→          error: 'File is too small to be a valid PDF',
   402→          errorCode: 'SIZE_TOO_SMALL',
   403→        }
   404→        onProgress?.([...progresses])
   405→        continue
   406→      }
   407→    }
   408→
   409→    // ── Process files with concurrent uploads ─────────────────────────────
   410→    const filesToUpload = files.filter((_, i) => progresses[i].status !== 'error')
   411→    const results = new Map<number, { file?: PdfFile; error?: any }>()
   412→
   413→    // Process with limited concurrency
   414→    const queue = [...filesToUpload.map((f, idx) => {
   415→      const originalIndex = files.indexOf(f)
   416→      return { file: f, originalIndex }
   417→    })]
   418→
   419→    const processFile = async (item: { file: File; originalIndex: number }) => {
   420→      const { file, originalIndex } = item
   421→      const isLargeFile = file.size > UPLOAD_LIMITS.CHUNK_THRESHOLD
   422→
   423→      try {
   424→        // Phase 1: Client-side hash for large files (pre-dup check)
   425→        let clientHash: string | undefined
   426→        if (isLargeFile) {
   427→          progresses[originalIndex] = {
   428→            ...progresses[originalIndex],
   429→            progress: 5,
   430→            status: 'validating' as const,
   431→          }
   432→          onProgress?.([...progresses])
   433→
   434→          clientHash = await computeClientHash(file)
   435→
   436→          // Pre-flight duplicate check for large files
   437→          const checkResponse = await fetch(`/api/files/check-duplicate?hash=${clientHash}`)
   438→          if (checkResponse.ok) {
   439→            const checkData = await checkResponse.json()
   440→            if (checkData.isDuplicate) {
   441→              progresses[originalIndex] = {
   442→                ...progresses[originalIndex],
   443→                progress: 100,
   444→                status: 'duplicate' as const,
   445→                duplicateInfo: checkData.existingFile,
   446→              }
   447→              onProgress?.([...progresses])
   448→              results.set(originalIndex, { error: { code: 'DUPLICATE_FILE', duplicate: checkData } })
   449→              return
   450→            }
   451→          }
   452→        }
   453→
   454→        // Phase 2: Upload
   455→        progresses[originalIndex] = {
   456→          ...progresses[originalIndex],
   457→          progress: isLargeFile ? 10 : 0,
   458→          status: 'uploading' as const,
   459→        }
   460→        onProgress?.([...progresses])
   461→
   462→        let data: any
   463→
   464→        if (isLargeFile && clientHash) {
   465→          // Chunked upload for large files
   466→          const result = await chunkedUploadWithProgress(file, clientHash, (p) => {
   467→            // Map chunk progress to 10-95 range (leave 5% for post-processing)
   468→            const mappedProgress = 10 + Math.round(p * 0.85)
   469→            progresses[originalIndex] = {
   470→              ...progresses[originalIndex],
   471→              progress: mappedProgress,
   472→            }
   473→            onProgress?.([...progresses])
   474→          })
   475→          data = result.data
   476→        } else {
   477→          // Standard upload with real progress
   478→          const result = await uploadWithProgress(file, (p) => {
   479→            // Map XHR progress to 0-95 range
   480→            const mappedProgress = Math.round(p * 0.95)
   481→            progresses[originalIndex] = {
   482→              ...progresses[originalIndex],
   483→              progress: mappedProgress,
   484→            }
   485→            onProgress?.([...progresses])
   486→          })
   487→
   488→          if (result.response.status === 409) {
   489→            // Duplicate detected
   490→            progresses[originalIndex] = {
   491→              ...progresses[originalIndex],
   492→              progress: 100,
   493→              status: 'duplicate' as const,
   494→              duplicateInfo: result.data.duplicate?.existingFile,
   495→            }
   496→            onProgress?.([...progresses])
   497→            results.set(originalIndex, { error: { code: 'DUPLICATE_FILE', duplicate: result.data.duplicate } })
   498→            return
   499→          }
   500→
   501→          if (result.response.status === 422) {
   502→            // Virus detected
   503→            progresses[originalIndex] = {
   504→              ...progresses[originalIndex],
   505→              progress: 100,
   506→              status: 'error' as const,
   507→              error: result.data.error || 'Security threat detected',
   508→              errorCode: 'VIRUS_DETECTED',
   509→            }
   510→            onProgress?.([...progresses])
   511→            results.set(originalIndex, { error: { code: 'VIRUS_DETECTED' } })
   512→            return
   513→          }
   514→
   515→          if (!result.response.ok) {
   516→            const errorMsg = result.data?.error || 'Upload failed'
   517→            const errorCode = result.data?.code || 'UPLOAD_FAILED'
   518→            throw Object.assign(new Error(errorMsg), { code: errorCode })
   519→          }
   520→
   521→          data = result.data
   522→        }
   523→
   524→        // Phase 3: Success
   525→        progresses[originalIndex] = {
   526→          ...progresses[originalIndex],
   527→          progress: 100,
   528→          status: 'success',
   529→          warnings: data.warnings,
   530→        }
   531→        onProgress?.([...progresses])
   532→
   533→        results.set(originalIndex, { file: data.file })
   534→      } catch (error: any) {
   535→        const errorCode = error.code || 'UPLOAD_FAILED'
   536→        const errorMessage = error.message || 'Upload failed'
   537→
   538→        // Check if this was a duplicate from chunked upload
   539→        if (errorCode === 'DUPLICATE_FILE' && error.duplicate) {
   540→          progresses[originalIndex] = {
   541→            ...progresses[originalIndex],
   542→            progress: 100,
   543→            status: 'duplicate' as const,
   544→            duplicateInfo: error.duplicate.existingFile,
   545→          }
   546→          onProgress?.([...progresses])
   547→          results.set(originalIndex, { error: { code: 'DUPLICATE_FILE', duplicate: error.duplicate } })
   548→          return
   549→        }
   550→
   551→        progresses[originalIndex] = {
   552→          ...progresses[originalIndex],
   553→          progress: 100,
   554→          status: 'error',
   555→          error: errorMessage,
   556→          errorCode,
   557→        }
   558→        onProgress?.([...progresses])
   559→        results.set(originalIndex, { error })
   560→      }
   561→    }
   562→
   563→    // Run with limited concurrency
   564→    const executing: Promise<void>[] = []
   565→    for (const item of queue) {
   566→      const p = processFile(item).then(() => {
   567→        executing.splice(executing.indexOf(p), 1)
   568→      })
   569→      executing.push(p)
   570→
   571→      if (executing.length >= UPLOAD_LIMITS.MAX_CONCURRENT) {
   572→        await Promise.race(executing)
   573→      }
   574→    }
   575→    await Promise.all(executing)
   576→
   577→    // Collect successful uploads
   578→    for (const [_, result] of results) {
   579→      if (result.file) {
   580→        uploadedFiles.push(result.file)
   581→      }
   582→    }
   583→
   584→    // Refresh file list after upload
   585→    if (uploadedFiles.length > 0) {
   586→      await get().fetchFiles()
   587→    }
   588→
   589→    return uploadedFiles
   590→  },
   591→
   592→  toggleStar: async (id: string) => {
   593→    const file = get().recentFiles.find((f) => f.id === id)
   594→    if (!file) return
   595→
   596→    // Optimistic update
   597→    set({
   598→      recentFiles: get().recentFiles.map((f) =>
   599→        f.id === id ? { ...f, starred: !f.starred } : f
   600→      ),
   601→    })
   602→
   603→    try {
   604→      const response = await fetch(`/api/files/${id}`, {
   605→        method: 'PATCH',
   606→        headers: { 'Content-Type': 'application/json' },
   607→        body: JSON.stringify({ starred: !file.starred }),
   608→      })
   609→
