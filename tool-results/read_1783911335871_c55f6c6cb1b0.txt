   118→function BatchCompressView({ onBack, addHistory }: { onBack: () => void; addHistory: (summary: string, details?: Record<string, any>, status?: 'success' | 'error' | 'partial') => void }) {
   119→  const { recentFiles, batchCompress, isCompressing, compressionProgress } = useAppStore()
   120→  const { toast } = useToast()
   121→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   122→  const [preset, setPreset] = useState<CompressionPreset>('balanced')
   123→  const [results, setResults] = useState<any>(null)
   124→
   125→  const toggleFile = (id: string) => {
   126→    setSelectedIds((prev) => {
   127→      const next = new Set(prev)
   128→      if (next.has(id)) next.delete(id)
   129→      else next.add(id)
   130→      return next
   131→    })
   132→  }
   133→
   134→  const handleBatchCompress = async () => {
   135→    if (selectedIds.size === 0) return
   136→    const result = await batchCompress(Array.from(selectedIds), preset)
   137→    if (result) {
   138→      setResults(result)
   139→      addHistory(
   140→        `Batch compressed ${result.summary.success} files — saved ${formatFileSize(result.summary.totalSavedBytes)}`,
   141→        {
   142→          fileCount: result.summary.success,
   143→          totalSavedBytes: result.summary.totalSavedBytes,
   144→          errors: result.summary.errors,
   145→          preset: preset,
   146→        },
   147→        result.summary.errors > 0 ? 'partial' : 'success'
   148→      )
   149→      toast({
   150→        title: 'Batch Compression Complete',
   151→        description: `${result.summary.success} files compressed, saved ${formatFileSize(result.summary.totalSavedBytes)}`,
   152→      })
   153→    }
   154→  }
   155→
   156→  return (
   157→    <div className="space-y-4">
   158→      <div className="flex items-center justify-between">
   159→        <div>
   160→          <h3 className="text-sm font-semibold text-gray-900">Batch Compress</h3>
   161→          <p className="text-xs text-gray-400">Select multiple files to compress at once</p>
   162→        </div>
   163→        <div className="flex items-center gap-2">
   164→          <Button variant="outline" size="sm" className="text-xs" onClick={onBack}>
   165→            Single File Mode
   166→          </Button>
   167→        </div>
   168→      </div>
   169→
   170→      {/* Preset selector */}
   171→      <div className="grid grid-cols-3 gap-2">
   172→        {PRESET_CARDS.map((p) => (
   173→          <button
   174→            key={p.id}
   175→            onClick={() => setPreset(p.id)}
   176→            className={cn(
   177→              'p-3 rounded-lg border-2 text-left transition-all',
   178→              preset === p.id ? 'border-[#4A90D9] shadow-sm' : 'border-gray-100 hover:border-gray-200'
   179→            )}
   180→          >
   181→            <p className="text-xs font-medium text-gray-800">{p.label}</p>
   182→            <p className="text-[10px] text-gray-400 mt-0.5">{p.quality} · {p.savings}</p>
   183→          </button>
   184→        ))}
   185→      </div>
   186→
   187→      {/* File list */}
   188→      <div className="space-y-1.5 max-h-64 overflow-y-auto">
   189→        {recentFiles.map((file) => (
   190→          <button
   191→            key={file.id}
   192→            onClick={() => toggleFile(file.id)}
   193→            className={cn(
   194→              'w-full flex items-center gap-3 p-3 rounded-lg border transition-all text-left',
   195→              selectedIds.has(file.id) ? 'border-[#4A90D9] bg-[#4A90D9]/5' : 'border-gray-100 hover:border-gray-200'
   196→            )}
   197→          >
   198→            <div className={cn(
   199→              'w-4 h-4 rounded border-2 flex items-center justify-center shrink-0',
   200→              selectedIds.has(file.id) ? 'border-[#4A90D9] bg-[#4A90D9]' : 'border-gray-300'
   201→            )}>
   202→              {selectedIds.has(file.id) && <CheckCircle2 className="w-3 h-3 text-white" />}
   203→            </div>
   204→            <FileText className="w-4 h-4 text-gray-400 shrink-0" />
   205→            <div className="flex-1 min-w-0">
   206→              <p className="text-xs font-medium text-gray-700 truncate">{file.name}</p>
   207→            </div>
   208→            <span className="text-[10px] text-gray-400 shrink-0">{formatFileSize(file.size)}</span>
   209→          </button>
   210→        ))}
   211→      </div>
   212→
   213→      {results ? (
   214→        <Card className="border-green-200 bg-green-50/50">
   215→          <CardContent className="p-4">
   216→            <div className="flex items-center gap-2 mb-2">
   217→              <CheckCircle2 className="w-4 h-4 text-green-600" />
   218→              <span className="text-sm font-medium text-green-800">Batch Complete</span>
   219→            </div>
   220→            <div className="grid grid-cols-3 gap-3 text-center">
   221→              <div>
   222→                <p className="text-lg font-bold text-green-600">{results.summary.success}</p>
   223→                <p className="text-[10px] text-gray-500">Compressed</p>
   224→              </div>
   225→              <div>
   226→                <p className="text-lg font-bold text-gray-800">{formatFileSize(results.summary.totalSavedBytes)}</p>
   227→                <p className="text-[10px] text-gray-500">Total Saved</p>
   228→              </div>
   229→              <div>
   230→                <p className="text-lg font-bold text-red-500">{results.summary.errors}</p>
   231→                <p className="text-[10px] text-gray-500">Failed</p>
   232→              </div>
   233→            </div>
   234→            <Button variant="outline" size="sm" className="w-full mt-3 text-xs" onClick={() => setResults(null)}>
   235→              Compress More
   236→            </Button>
   237→          </CardContent>
   238→        </Card>
   239→      ) : (
   240→        <Button
   241→          className="w-full bg-green-600 hover:bg-green-700 text-white"
   242→          disabled={selectedIds.size === 0 || isCompressing}
   243→          onClick={handleBatchCompress}
   244→        >
   245→          {isCompressing ? (
   246→            <>
   247→              <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   248→              Compressing {compressionProgress?.current || 0}/{compressionProgress?.total || selectedIds.size}...
   249→            </>
   250→          ) : (
   251→            <>
   252→              <ArrowDown className="w-4 h-4 mr-2" />
   253→              Compress {selectedIds.size} File{selectedIds.size !== 1 ? 's' : ''}
   254→            </>
   255→          )}
   256→        </Button>
   257→      )}
   258→    </div>
   259→  )
   260→}

   264→export function CompressPdf() {
   265→  const {
   266→    recentFiles,
   267→    compressFile,
   268→    compressFilePreview,
   269→    isCompressing,
   270→    compressionProgress,
   271→    setCurrentView,
   272→  } = useAppStore()
   273→
   274→  const { toast } = useToast()
   275→  const { history, addHistory, deleteItem, clearHistory, isLoaded } = useToolHistory('compress', 'Compress PDF')
   276→
   277→  const [selectedFile, setSelectedFile] = useState<PdfFile | null>(null)
   278→  const [compressionPreset, setCompressionPreset] = useState<CompressionPreset>('balanced')
   279→  const [result, setResult] = useState<CompressionResult | null>(null)
   280→  const [preview, setPreview] = useState<PreviewData | null>(null)
   281→  const [isLoadingPreview, setIsLoadingPreview] = useState(false)
   282→  const [isBatchMode, setIsBatchMode] = useState(false)
   283→
   284→  // Fetch preview when file changes
   285→  useEffect(() => {
   286→    if (!selectedFile) return
   287→
   288→    let cancelled = false
   289→    const fetchPreview = async () => {
   290→      setIsLoadingPreview(true)
   291→      const data = await compressFilePreview(selectedFile.id)
   292→      if (!cancelled) {
   293→        setPreview(data || null)
   294→        setIsLoadingPreview(false)
   295→      }
   296→    }
   297→
   298→    fetchPreview()
   299→    return () => { cancelled = true }
   300→  }, [selectedFile?.id, compressFilePreview])
   301→
   302→  const handleCompress = useCallback(async () => {
   303→    if (!selectedFile) return
   304→
   305→    setResult(null)
   306→    const response = await compressFile(selectedFile.id, compressionPreset)
   307→
   308→    if (response) {
   309→      setResult(response)
   310→      addHistory(
   311→        `Compressed ${selectedFile.name} — saved ${response.compression.savedPercent}%`,
   312→        {
   313→          fileName: selectedFile.name,
   314→          fileSize: selectedFile.size,
   315→          originalSize: response.compression.originalSize,
   316→          compressedSize: response.compression.compressedSize,
   317→          savedPercent: response.compression.savedPercent,
   318→          savedBytes: response.compression.savedBytes,
   319→          preset: compressionPreset,
   320→          pages: selectedFile.pages,
   321→        },
   322→        'success'
   323→      )
   324→      toast({
   325→        title: 'Compression Complete',
   326→        description: `Saved ${response.compression.savedPercent}% — ${formatFileSize(response.compression.savedBytes)} reduced in ${response.durationMs ? (response.durationMs / 1000).toFixed(1) + 's' : 'a moment'}`,
   327→      })
   328→    } else {
   329→      toast({
   330→        title: 'Compression Failed',
   331→        description: 'Failed to compress the PDF file. Please try again.',
   332→        variant: 'destructive',
   333→      })
   334→    }
   335→  }, [selectedFile, compressionPreset, compressFile, toast, addHistory])
   336→
   337→  const handleDownload = useCallback(() => {
   338→    if (!result) return
   339→    fetch(`/api/files/${result.file.id}/download?download=1`)
   340→      .then((response) => {
   341→        if (!response.ok) throw new Error('Download failed')
   342→        return response.blob()
   343→      })
   344→      .then((blob) => {
   345→        const url = window.URL.createObjectURL(blob)
   346→        const a = document.createElement('a')
   347→        a.href = url
   348→        a.download = result.file.name
   349→        document.body.appendChild(a)
   350→        a.click()
   351→        document.body.removeChild(a)
   352→        window.URL.revokeObjectURL(url)
   353→      })
   354→      .catch((err) => {
   355→        console.error('Download error:', err)
   356→        toast({ title: 'Download Failed', variant: 'destructive' })
   357→      })
   358→
   359→    toast({ title: 'Download Started', description: `Downloading ${result.file.name}` })
   360→  }, [result, toast])
   361→
   362→  const savedPercentNum = result ? parseFloat(result.compression.savedPercent) : 0
   363→  const compressionRatio = result ? (result.compression.compressedSize / result.compression.originalSize) * 100 : 100
   364→
   365→  if (isBatchMode) {
   366→    return (
   367→      <div className="h-full flex flex-col">
   368→        <div className="flex items-center justify-between px-4 md:px-6 py-3 md:py-4 border-b border-gray-100 bg-white">
   369→          <div className="flex items-center gap-3">
   370→            <div className="w-8 h-8 bg-green-50 rounded-lg flex items-center justify-center">
   371→              <FileDown className="w-4 h-4 text-green-600" />
   372→            </div>
   373→            <div>
   374→              <h1 className="text-base md:text-lg font-semibold text-gray-800">Batch Compress PDFs</h1>
   375→              <p className="text-xs text-gray-400">Reduce multiple files at once</p>
   376→            </div>
   377→          </div>
   378→          <button onClick={() => setCurrentView('home')} className="p-2 rounded-md hover:bg-gray-100 text-gray-400 hover:text-gray-600 transition-colors">
   379→            <X className="w-5 h-5" />
   380→          </button>
   381→        </div>
   382→        <div className="p-4 md:p-6 md:flex-1 md:min-h-0 md:overflow-auto pb-4 md:pb-0">
   383→          <BatchCompressView onBack={() => setIsBatchMode(false)} addHistory={addHistory} />
   384→        </div>
   385→      </div>
   386→    )
   387→  }
   388→
   389→  return (
   390→    <div className="h-full flex flex-col">
   391→      {/* Header */}
   392→      <div className="flex items-center justify-between px-4 md:px-6 py-3 md:py-4 border-b border-gray-100 bg-white">
   393→        <div className="flex items-center gap-3">
   394→          <div className="w-8 h-8 bg-green-50 rounded-lg flex items-center justify-center">
   395→            <FileDown className="w-4 h-4 text-green-600" />
   396→          </div>
   397→          <div>
   398→            <h1 className="text-base md:text-lg font-semibold text-gray-800">Compress PDF</h1>
   399→            <p className="text-xs text-gray-400">Reduce PDF file size for easier sharing</p>
   400→          </div>
   401→        </div>
   402→        <div className="flex items-center gap-2">
   403→          <Button variant="outline" size="sm" className="text-xs" onClick={() => setIsBatchMode(true)}>
   404→            <Layers className="w-3.5 h-3.5 mr-1" />
   405→            Batch Mode
   406→          </Button>
   407→          <button onClick={() => setCurrentView('home')} className="p-2 rounded-md hover:bg-gray-100 text-gray-400 hover:text-gray-600 transition-colors">
   408→            <X className="w-5 h-5" />
   409→          </button>
   410→        </div>
   411→      </div>
   412→
   413→      <div className="flex-1 flex flex-col md:flex-row min-h-0 overflow-y-auto md:overflow-y-hidden">
   414→        {/* Main Content */}
   415→        <div className="p-4 md:p-6 md:flex-1 md:min-h-0 md:overflow-auto pb-4 md:pb-0">
   416→          {!selectedFile ? (
   417→            /* File Selection */
   418→            <div>
   419→              <h3 className="text-sm font-medium text-gray-600 mb-3">Select a file to compress</h3>
   420→              {recentFiles.length > 0 ? (
   421→                <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   422→                  {recentFiles.map((file) => (
   423→                    <button
   424→                      key={file.id}
   425→                      onClick={() => { setSelectedFile(file); setResult(null) }}
   426→                      className="flex items-start gap-3 p-4 bg-white rounded-lg border border-gray-100 hover:border-green-400 hover:shadow-md transition-all text-left group"
   427→                    >
   428→                      <div className="w-10 h-10 bg-green-50 rounded-lg flex items-center justify-center shrink-0">
   429→                        <FileText className="w-5 h-5 text-green-600" />
   430→                      </div>
   431→                      <div className="min-w-0 flex-1">
   432→                        <div className="text-sm font-medium text-gray-800 group-hover:text-green-600 truncate">{file.name}</div>

   440→                <div className="flex flex-col items-center justify-center py-16 text-gray-400 bg-white rounded-lg border border-dashed border-gray-300">
   441→                  <FileDown className="w-12 h-12 mb-3 text-gray-300" />
   442→                  <p className="text-sm mb-2">No files available</p>
   443→                  <p className="text-xs text-gray-300">Upload a PDF file first to compress it</p>
   444→                </div>
   445→              )}
   446→            </div>
   447→          ) : (
   448→            <div className="space-y-4">
   449→              {/* Selected File Info */}
   450→              <Card className="border-gray-200">
   451→                <CardContent className="p-4">
   452→                  <div className="flex items-center justify-between">
   453→                    <div className="flex items-center gap-3">
   454→                      <div className="w-10 h-10 bg-green-50 rounded-lg flex items-center justify-center shrink-0">
   455→                        <FileText className="w-5 h-5 text-green-600" />
   456→                      </div>
   457→                      <div>
   458→                        <div className="text-sm font-medium text-gray-800">{selectedFile.name}</div>
   459→                        <div className="text-xs text-gray-400 mt-0.5">
   460→                          {formatFileSize(selectedFile.size)} · {selectedFile.pages} page{selectedFile.pages !== 1 ? 's' : ''}
   461→                        </div>
   462→                      </div>

   464→                    <Button variant="ghost" size="sm" className="text-xs text-gray-500 hover:text-gray-700" onClick={() => { setSelectedFile(null); setResult(null); setPreview(null) }}>

   467→                  </div>
   468→                </CardContent>
   469→              </Card>
   470→
   471→              {/* Compression Preview */}
   472→              {preview && !result && (
   473→                <Card className="border-[#4A90D9]/20 bg-[#4A90D9]/5">

   496→                        )}
   497→                        {preview.preview.analysis.hasMetadata && (
   498→                          <div className="flex items-center gap-1.5 text-[11px] text-gray-500">

   503→                        <div className="flex items-center gap-1.5 text-[11px] text-gray-500">

   512→
   513→              {isLoadingPreview && !preview && !result && (
   514→                <Card className="border-gray-200">
   515→                  <CardContent className="p-4 flex items-center gap-2">
   516→                    <Loader2 className="w-4 h-4 text-[#4A90D9] animate-spin" />
   517→                    <span className="text-xs text-gray-500">Analyzing compression potential...</span>
